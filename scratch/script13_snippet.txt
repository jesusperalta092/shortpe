refreshEpisodeSourceForPrefetchIfStale → URL basi selalu ke-refresh.
        return urlHasTimeToken(currentSourceUrl)
            ? BROWSER_PREFETCH_NEIGHBOR_REFRESH_TTL_MS
            : BROWSER_PREFETCH_FORCE_REFRESH_TTL_MS;
    }

    function clearMediaPreloadEntry(key) {
        const entry = mediaPreloadState.get(key);
        if (!entry) return;
        const video = entry.video;
        const hlsInstance = entry.hls;
        if (hlsInstance && typeof hlsInstance.destroy === 'function') {
            try { hlsInstance.destroy(); } catch (e) { /* ignore */ }
        }
        if (video) {
            try { video.pause(); } catch (e) { /* ignore */ }
            try { video.removeAttribute('src'); } catch (e) { /* ignore */ }
            try { video.load(); } catch (e) { /* ignore */ }
            try { video.remove(); } catch (e) { /* ignore */ }
        }
        mediaPreloadState.delete(key);
    }

    function createHiddenPreloadVideo() {
        const video = document.createElement('video');
        video.muted = true;
        video.preload = 'auto';
        video.playsInline = true;
        enforceVideoNoReferrer(video);
        video.disablePictureInPicture = true;
        video.setAttribute('aria-hidden', 'true');
        video.setAttribute('tabindex', '-1');
        video.style.position = 'absolute';
        video.style.width = '1px';
        video.style.height = '1px';
        video.style.opacity = '0';
        video.style.pointerEvents = 'none';
        video.style.left = '-9999px';
        video.style.top = '-9999px';
        stage.appendChild(video);
        return video;
    }

    function cleanupOutOfWindowPreloads(currentEp) {
        const activeWindow = new Set();
        for (let offset = 1; offset <= BROWSER_PREFETCH_RANGE; offset += 1) {
            activeWindow.add(String(currentEp + offset));
            activeWindow.add(String(currentEp - offset));
        }
        for (const key of mediaPreloadState.keys()) {
            if (!activeWindow.has(String(key))) {
                clearMediaPreloadEntry(key);
            }
        }
    }
